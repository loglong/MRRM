From prisma/schema.prisma (USE THESE ENUMS):
```prisma
model Demand {
  id          String   @id @default(uuid())
  patientId   String
  patient     Patient @relation(fields: [patientId], references: [id])
  type        DemandType
  title       String
  description String
  priority    DemandPriority @default(MEDIUM)
  status      DemandStatus @default(OPEN)
  orgId       String
  source      DemandSource @default(WALK_IN)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  closedAt    DateTime?
  deletedAt   DateTime?

  statusHistory DemandStatusHistory[]

  @@index([patientId])
  @@index([orgId, status])
  @@index([orgId, createdAt])
}

enum DemandType {
  CONSULTATION
  TREATMENT
  FOLLOWUP
  OTHER
}

enum DemandPriority {
  LOW
  MEDIUM
  HIGH
  URGENT
}

enum DemandStatus {
  OPEN           // 新建 - initial status (plan said: NEW)
  IN_PROGRESS    // 跟进中 - same
  PENDING        // 待处理 - plan said: CONVERTED (meaning patient agreed)
  FULFILLED      // 已成交 - plan said: COMPLETED
  CANCELLED      // 已取消
  LOST           // 流失
}

enum DemandSource {
  PHONE
  WECHAT
  WEB
  WALK_IN
  REFERRAL
  CAMPAIGN
  OTHER
}

model DemandStatusHistory {
  id         String   @id @default(uuid())
  demandId   String
  demand     Demand @relation(fields: [demandId], references: [id], onDelete: Cascade)
  fromStatus DemandStatus?
  toStatus   DemandStatus
  changedBy  String
  changedAt  DateTime @default(now())
  notes      String?

  @@index([demandId])
}
```
